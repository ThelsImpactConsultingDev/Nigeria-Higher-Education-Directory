#!/usr/bin/env python3 """Gate the build. Exits non-zero if the data is not fit to publish. python scripts/validate.py [--data build/data.json] [--source data/source.xlsx] ERRORS stop the build, so the live site keeps serving the last good version. WARNINGS are printed for you to fix at source but do not block publishing. """ import argparse, json, os, sys from collections import Counter, defaultdict sys.path.insert(0, os.path.dirname(os.path.abspath(__file__))) from clean import (SHEETS, COMING_SOON, ZONES, STATE_ZONE, OWNERS, MIN_EXPECTED_ROWS, norm_state, zone_for, norm_inst) errors, warnings = [], [] def err(m): errors.append(m) def warn(m): warnings.append(m) def expand(pack): """Rebuild flat rows from the compact payload, mirroring the page's JS.""" if not pack: return [] out = [] for row in pack['r']: inst = pack['i'][row[0]] out.append(dict(institution=inst[0], state=inst[1], zone=inst[2], website=inst[3], proprietorship=inst[4], unit=pack['u'][row[1]], programme=pack['p'][row[2]], programme_type=pack['t'][row[3]])) return out def check_payload(payload): for key in list(SHEETS) + COMING_SOON: if key not in payload: err(f"payload is missing the '{key}' category entirely") for key in COMING_SOON: if payload.get(key) not in (None, {}): err(f"'{key}' should be null (coming soon) but carries data") rows = [] for key in SHEETS: r = expand(payload.get(key)) if not r: err(f"category '{key}' has no rows — the sheet may have been renamed or emptied") rows.extend(r) if len(rows) < MIN_EXPECTED_ROWS: err(f"only {len(rows):,} rows built, expected at least {MIN_EXPECTED_ROWS:,}. " f"This usually means a truncated or partly-synced export.") bad_field = bad_zone = bad_state = bad_owner = 0 no_web = 0 inst_states = defaultdict(set) for r in rows: if not all((r['institution'], r['unit'], r['programme'], r['programme_type'])): bad_field += 1 if r['zone'] not in ZONES: bad_zone += 1 if (r['state'] or '').lower() not in STATE_ZONE: bad_state += 1 if r['proprietorship'] not in OWNERS: bad_owner += 1 if not r['website']: no_web += 1 inst_states[r['institution']].add(r['state']) if bad_field: err(f"{bad_field:,} rows are missing a required field") if bad_zone: err(f"{bad_zone:,} rows carry a zone outside the six geo-political zones") if bad_state: err(f"{bad_state:,} rows carry a state not in the 36-states-plus-FCT list") if bad_owner: err(f"{bad_owner:,} rows have an ownership value outside {OWNERS}") # No invented values may reappear. for key in SHEETS: pack = payload.get(key) or {} if 'a' in pack: err(f"'{key}' carries an accreditation column — placeholder data has returned") if pack.get('r') and len(pack['r'][0]) != 4: err(f"'{key}' rows have {len(pack['r'][0])} columns, expected 4 " f"(extra columns usually mean cut-off marks crept back in)") split = Counter(r['institution'] for r in rows) multi = {i: s for i, s in inst_states.items() if len(s) > 1} if multi: warn(f"{len(multi)} institution name(s) appear in more than one state — " f"possible duplicate naming: " + ", ".join(list(multi)[:5])) if no_web: warn(f"{no_web:,} rows ({no_web*100//max(1,len(rows))}%) have no website in the source") return rows def check_source_zones(path): """Compare the workbook's own zone column against the derived zone. Disagreements are the source's problem, not ours — warn, don't block.""" if not os.path.exists(path): return try: import pandas as pd xl = pd.ExcelFile(path) except Exception as e: warn(f"could not re-open source for the zone cross-check: {e}") return zone_cols = ['Geo-Political Zones', 'GEO-POLITICAL ZONES'] mismatches = Counter() for key, cfg in SHEETS.items(): if cfg['sheet'] not in xl.sheet_names: continue df = xl.parse(cfg['sheet']) df.columns = [c.strip() for c in df.columns] zc = next((c for c in zone_cols if c in df.columns), None) if not zc: continue for _, r in df.iterrows(): st = norm_state(r.get(cfg['state'])) derived = zone_for(st) stated = str(r.get(zc) or '').replace('_', ' ').strip().title() if derived and stated in ZONES and stated != derived: inst = norm_inst(r.get(cfg['institution'])) or '(unnamed)' mismatches[(inst, st, stated, derived)] += 1 if mismatches: warn(f"{sum(mismatches.values())} row(s) where the workbook's zone column " f"disagrees with the state. The build used the state-derived zone. " f"Worth correcting in the spreadsheet:") for (inst, st, stated, derived), n in mismatches.most_common(10): warnings.append(f" {inst} ({st}): sheet says {stated}, should be {derived} [{n} rows]") def main(): ap = argparse.ArgumentParser() ap.add_argument('--data', default='build/data.json') ap.add_argument('--source', default='data/source.xlsx') args = ap.parse_args() with open(args.data, encoding='utf-8') as f: payload = json.load(f) rows = check_payload(payload) check_source_zones(args.source) print("=" * 62) print(f"{'VALIDATION':<40}") print("=" * 62) if rows: zc = Counter(r['zone'] for r in rows) print(f"listings : {len(rows):,}") print(f"institutions : {len(set(r['institution'] for r in rows)):,}") print(f"states : {len(set(r['state'] for r in rows))}") for z in ZONES: print(f" {z:<15} {zc[z]:>7,}") if warnings: print("\nWARNINGS (site still publishes)") for w in warnings: print(f" ! {w}") if errors: print("\nERRORS (build stopped, live site unchanged)") for e in errors: print(f" x {e}") print("\nFIX THE SPREADSHEET AND RE-RUN.") sys.exit(1) print("\nAll checks passed — safe to publish.") if __name__ == '__main__': main()
